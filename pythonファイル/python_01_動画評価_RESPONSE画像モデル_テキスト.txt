def run_responses_session(model, SP_INPUT, INPUT):
    url = "https://api.openai.com/v1/responses"
    message = INPUT + SP_INPUT
    if model == "chatgpt-4o-latest":
        payload_response = {
            "model": model,
            "temperature": 0,
            "max_output_tokens": 800,
            "input": [
                {
                    "type": "message",
                    "role": "user", "content": message}]
        }
    if model == "gpt-5" or model == "gpt-5-nano" or model == "gpt-5-mini":
        payload_response = {
            "model": model,
            "max_output_tokens": 2000,
            "input": [
                {
                    "type": "message",
                    "role": "user", "content": message}],
            "reasoning": {"effort": "low"}
        }
    else:
        payload_response = {
            "model": model,
            "max_output_tokens": 1200,
            "input": [
                {
                    "type": "message",
                    "role": "user", "content": message}]
        }

    try:
        response = requests.post(url, headers=headers, json=payload_response,
                                 timeout=(5, 360))
        response.raise_for_status()

    except requests.exceptions.ReadTimeout:
        return "Time-Out"

    except requests.exceptions.RequestException as e:
        # logging.warning(f"⚠️non-200 status code: {response.status_code}, status: {response.status_code}, body: , {response.text[:4000]} ")
        if response.status_code == 429:
            return 4290
        else:
            return "Not 200"

    return response.json()

def normalize_col_name(x):
    """
    列名をどんな形式からでも「純粋な数字文字列」に正規化する。
    """
    if isinstance(x, int):
        return str(x)

    if not isinstance(x, str):
        x = str(x)

    # 前後の空白・タブ・不可視文字を除去
    x = x.strip()

    # BOM 除去
    x = x.replace("\ufeff", "")

    # 数字以外を除去（安全策）
    x = "".join(ch for ch in x if ch.isdigit())

    return x


def needs_re_evaluation(csv_path: str, vid_idx: int) -> bool:
    df = pd.read_csv(csv_path, index_col=0, encoding="shift_jis")

    normalized_cols = {normalize_col_name(c): c for c in df.columns}
    key = normalize_col_name(vid_idx)

    if key not in normalized_cols:
        return False

    col = normalized_cols[key]
    values = pd.to_numeric(df[col].iloc[0:34], errors="coerce").fillna(0)

    s = values.sum()

    # 正常パターン
    NORMAL = {0, 50, 60, 61}

    # 正常なら再評価しない
    if s in NORMAL:
        return False

    # それ以外は異常 → 再評価
    return True

# 11. responseモデル：感情カテゴリ・次元・単語の保存用
def extract_results_as_choice(result):
    # Responses API の output を探す
    outputs = result.get("output", [])
    for item in outputs:
        if not isinstance(item, dict):
            continue

        # content がある message ブロックを探す
        content = item.get("content")
        if not isinstance(content, list):
            continue

        # content の中に output_text があるか探す
        for c in content:
            if isinstance(c, dict) and c.get("type") == "output_text":
                text = c.get("text", "")
                # JSON 部分だけ抽出
                start = text.find("{")
                end = text.rfind("}")
                if start != -1 and end != -1:
                    try:
                        return json.loads(text[start:end+1])
                    except:
                        return None

    # 見つからなければ None
    return None

def fix_broken_json_array(text):
    """
    途中で切れた JSON 配列を安全に修復する。
    例: ["Abc", "Def", "Gh → ["Abc", "Def"]
    """

    # 配列の開始位置を探す
    start = text.find("[")
    if start == -1:
        return []

    # 配列部分を抽出
    arr = text[start:]

    # 末尾が ] で終わっていなければ閉じる
    if not arr.strip().endswith("]"):
        arr = arr.rstrip()  # 改行など除去
        arr += "]"

    # JSON としてパースを試みる
    try:
        return json.loads(arr)
    except Exception:
        # パース失敗 → 要素ごとに手動で修復
        # "xxx" の形を正規表現で抽出
        items = re.findall(r'"([^"]*)"', arr)

        # 最後の要素が途中で切れている場合を除去
        fixed = []
        for item in items:
            if "\n" in item or "\r" in item:
                continue
            fixed.append(item)

        return fixed


def repair_arguments_array(arg_str: str) -> str:
    """
    function_call.arguments の中の壊れた配列部分を修復する。
    例: "calmness","clarity",,"clarity-of-spiritual-
      → "calmness","clarity","clarity-of-spiritual-"
    """

    if not isinstance(arg_str, str):
        return arg_str

    s = arg_str.strip()

    # 1) よくある壊れ方：",," → "," に修正
    s = re.sub(r',\s*,', ',', s)

    # 2) words が配列でなく文字列扱いになっている壊れ方を修正
    #    {"words":"a","b","c"} → {"words":["a","b","c"]}
    if '"words":"' in s and not '"words":[' in s:
        s = s.replace('"words":"', '"words":["')
        # 最後の " を ] に置換（壊れている場合は後で補完）
        s = re.sub(r'"(\s*})', r'"]\1', s)

    # 3) 配列の途中で切れている場合、末尾の壊れた部分を削る
    s = re.sub(r'"([^"]*)$', r'"\1"', s)

    # 4) 配列の閉じ括弧が欠けている場合に補完
    if '"words":[' in s and not re.search(r'\]\s*}', s):
        # words の後ろに ]} を補完
        s = re.sub(r'\s*}$', ']}', s)

    return s

    if not raw_args1:
        error_dir = "./モデル別ファイル/論文用/json_parse_errors"
        os.makedirs(error_dir, exist_ok=True)

        # 保存ファイル名

        # result1 全体を保存（壊れた JSON も含めて）
        with open(save_path, "w", encoding="utf-8") as f:
            f.write(str(result1))

        print(f"Saved raw response to {save_path}")
        return None

    # JSON文字列ならdictに変換
    if isinstance(raw_args1, str):
        raw_args1 = json.loads(raw_args1)

    # emotions 部分の処理
    if isinstance(raw_args1, list):
        raw_args1 = raw_args1[0]

    vid_col = str(vid_idx)
    # 列が存在しなければ追加
    if vid_col not in df1.columns:
        df1[vid_col] = 0
    if vid_col not in df3.columns:
        df3[vid_col] = 1

    # print(raw_args1)
    emotions = raw_args1.get("emotions", [])

    # 順位ごとの点数を定義
    score_map = {0: 50, 1: 10, 2: 1}

    # emotions の順位に応じてスコアを付与
    # EMOTIONSにないものを"Other"に変換し、既存感情より後ろに配置
    valid = [e for e in emotions if e in EMOTIONS]
    if any(e not in EMOTIONS for e in emotions):
        others = ["Other"]
    else:
        others = []
    ordered_emotions = valid + others

    # 書き込み処理（ordered_emotionsの長さ分だけ）
    for rank, emo in enumerate(ordered_emotions):
        if emo in df1.index:
            df1.at[emo, vid_col] = score_map.get(rank, 0)

    # appraisals 部分の処理
    appraisals = raw_args1.get("appraisals", {})

    VALUES = []
    for app_name, value in appraisals.items():
        if app_name in df3.index:
            df3.at[app_name, vid_col] = value
            VALUES.append(value)
    now = datetime.now()
    if "usage" in result1:
        usage1 = result1["usage"]
        if "prompt_tokens" in usage1:
            token1 = usage1["prompt_tokens"]
        else:
            token1 = usage1["input_tokens"]
    print(now.strftime("%H:%M"), f"/ {vid_idx} / {model}", "★ EMO:", ordered_emotions,
          "★ APP:", VALUES, token1)

    # csvに保存
    df1.to_csv(file_path1, index=True, header=True)
    df3.to_csv(file_path3, index=True, header=True)
    if os.path.getsize(file_path1) == 0:
        raise RuntimeError(f"CSV書き込みに失敗しました: {file_path1}")
    if len(df1) < 34:
        raise RuntimeError(
            f"CSVが途中で欠落しています（{len(df1)} 行しかありません）: {file_path1}")
    if os.path.getsize(file_path3) == 0:
        raise RuntimeError(f"CSV書き込みに失敗しました: {file_path3}")
    if len(df3) < 14:
        raise RuntimeError(
            f"CSVが途中で欠落しています（{len(df3)} 行しかありません）: {file_path3}")


def fix_broken_json_array_text(text):
    """
    壊れた JSON 配列文字列を受け取り、
    修正済みの JSON 文字列を返す。

    例:
      '["Abc", "Def", "Gh'

# 14. resultの形の確認
def handle_api_result(result):
    # 1) 型チェック
    if not isinstance(result, dict):
        logging.error(
            f"❌ Unexpected API response type (not dict): {type(result)}")

        return None

    if "choices" in result and result["choices"]:
        choice = result["choices"][0]
        # Chat completions の場合、message は choice["message"]
        msg = choice.get("message", {})
        # 関数呼び出しが来ているか
        if "function_call" in msg:
            args_text = msg["function_call"].get("arguments", "")
            try:
                fixed_text = fix_broken_json_array_text(args_text)
                args = json.loads(fixed_text)
            except Exception:
                print("\n★ 評価：なし")
                return None
            return {"mode": "function_call", "args": args}
        # 通常のテキスト応答
        if "content" in msg:
            return {"mode": "message", "content": msg["content"]}
        # 互換性のため、choice に直接 'text' がある場合も扱う
        if "text" in choice:
            return {"mode": "text", "content": choice["text"]}

    # 3) output 形式（/v1/responses など）
    if "output" in result and result["output"]:
        # output は list のことが多い。中身を結合して扱う例：
        outputs = result["output"]
        # 例: outputs の中に {'type':'message','content':[{'type':'output_text','text':'...'}]} のような構造がある場合
        # ここでは簡易に text を抽出する汎用ロジックを使う
        texts = []
        for item in outputs:
            # item が dict で 'content' を持つ場合
            if isinstance(item, dict):
                # content が list の場合、その中の text を拾う
                cont = item.get("content")
                if isinstance(cont, list):
                    for c in cont:
                        if isinstance(c, dict) and "text" in c:
                            texts.append(c["text"])
                        elif isinstance(c, str):
                            texts.append(c)
                elif isinstance(cont, str):
                    texts.append(cont)
                # 直接 'text' キーがある場合
                elif "text" in item:
                    texts.append(item["text"])
            if item.get("type") == "output_text":
                content = item.get("content", [])
                texts = []
                for c in content:
                    if isinstance(c, dict) and "text" in c:
                        texts.append(c["text"])
                if texts:
                    return {"mode": "message", "content": "\n".join(texts)}
        joined = "\n".join(texts).strip()
        if joined:
            return {"mode": "message", "content": joined}
        # もし output に function_call 相当があるならその処理を追加（API仕様に依存）
        # ここでは見つからなければエラー扱いにする
    if item.get("type") == "reasoning":
        # reasoning には text が無いので、summary を返すか None にする
        summary = item.get("summary")
        if summary:
            texts.append("\n".join(summary))
        return {"mode": "message", "content": "\n".join(texts)}

    # 4) どちらの形式でもない場合はログ出力して None を返す
    logging.error("❌ Unexpected API response structure. keys=%s; full=%s", list(
        result.keys()), result)
    return None

def send_request(payload, vid_name, model):
    # OpenAIのLLMに送るresponseを作成

    URL_CHAT = "https://api.openai.com/v1/responses"

    try:
        response = requests.post(
            URL_CHAT,
            headers={"Authorization": f"Bearer {UN_API_KEY}",
                     "Content-Type": "application/json"},
            json=payload,
            timeout=(5, 360)
        )
        response.raise_for_status()
        return response

    except requests.exceptions.ReadTimeout:
        logging.warning("⌛️タイムアウト")

        error_log.append(f"{vid_name} / {model} / TIMEOUT")
        return None

    except requests.exceptions.RequestException as e:
        logging.warning(f"⚠️ リクエストエラー")

        error_log.append(f"{vid_name} / {model} / REQUEST")
        if "429" in str(e) and "Too Many Requests" in str(e):
            logging.warning("status code: 429 API使用制限 残量を追加してください。")
            return "Too-Many-Requests"
        else:
            logging.warning(e)
            return None

        except ValueError:
            logging.info(result1.text)
            return vid_idx
        except Exception as e:
            err = str(e)
            logging.error(err)
            return None

        if result1 != []:
            parsed1 = handle_api_result(result1)
            if parsed1 is None:
                return vid_idx
            # --- 既存の処理 ---
        emotion_word_appraisal(model, vid_idx,  result1, df21, df23,
                               new_path1, new_path3)

    except requests.HTTPError as http_err:
        logging.error(
            f"⚠️ HTTP error on {vid_name}: {http_err}")
        if http_err.response.status_code != 200 and http_err.response.status_code != 429:
            if response1:
                logging.info(response1.text)
            return vid_idx
        if http_err.response.status_code == 429:
            logging.warning(
                "\n★API使用制限★")
            return 0
    except Exception as e:
        logging.warning("Error:\n", e)
        err = getattr(e, 'response', None) or {}
        logging.error("Status: %s", getattr(err, "status_code", None))
        return vid_idx
    return vid_idx + 1
